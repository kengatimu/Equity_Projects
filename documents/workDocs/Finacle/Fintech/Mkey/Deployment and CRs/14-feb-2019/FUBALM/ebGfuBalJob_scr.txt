<--start
trace on
#+++++++++
#rep&Class
#+++++++++
        if(repexists("MKEY")==0)then
                createrep("MKEY")
        endif
        if(classexists("MKEY","GLBDATA")==0)then
                createclass("MKEY","GLBDATA",5)
        endif
#+++++++++++++
#fub Records
#+++++++++++++
        sv_s = "fuSolId | select distinct sol_id from tbaadm.gam where bank_id='99' and entity_cre_flg='Y' and del_flg='N' and acct_ownership<>'O' "
        sv_s = sv_s + " and exists (select 1 from tbaadm.ftx where bank_id='99' and acid = gam.acid and del_flg='N' ) "
        print(sv_s)
        sv_u = urhk_dbCursorOpenWithBind(sv_s)
        if(sv_u ==0)then
                MKEY.GLBDATA.curNum = BANCS.OUTPARAM.DB_CURSOR_NUMBER
        else
                print(" data fetch failed GAM ")
                goto ENDFUB
        endif
        sv_f = urhk_dbCursorFetch(MKEY.GLBDATA.curNum)
        if(sv_f !=0)then
                sv_f = urhk_dbCursorClose(MKEY.GLBDATA.curNum)
                print(sv_f)
                goto ENDFUB
        endif
        MKEY.GLBDATA.recNum = 0
        while(sv_f ==0)
                MKEY.GLBDATA.fuSolid = BANCS.OUTPARAM.fuSolId
                print(MKEY.GLBDATA.fuSolid)
                if(MKEY.GLBDATA.fuSolid != "")then
                        sv_u = "nohup exebatch babx4489 $B2K_SESSION_ID "  + MKEY.GLBDATA.fuSolid
                        print(sv_u)
                        sv_o = system(sv_u)
                        if(sv_o==0)then
                                print("fu bal success")
                                goto NXTREC
                        else
                                print("fu bal failed")
                                goto NXTREC
                        endif
                endif
                NXTREC:
                sv_f = urhk_dbCursorFetch(MKEY.GLBDATA.curNum)
                MKEY.GLBDATA.recNum = cint(MKEY.GLBDATA.recNum) + 1
        do
        sv_f = urhk_dbCursorClose(MKEY.GLBDATA.curNum)
        print(sv_f)
#+++++++
#clean up
#+++++++
ENDFUB:
        if(classexists("MKEY","GLBDATA")==1)then
                deleteclass("MKEY","GLBDATA")
        endif
        if(repexists("MKEY")==1)then
                deleterep("MKEY")
        endif
        exitscript
trace off
